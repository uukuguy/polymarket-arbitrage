       ```python
       """DNS baseline probe — Fly support ticket evidence collection.

       Phase 03.1 D-01 C: deploy this as a cron inside polyarb-l1 Fly machine to
       collect node-level baseline data on the EAI_NODATA / EAI_AGAIN chronic
       DNS failures (Sentry issue 121111789). Outputs JSON-lines to stdout
       (one per probe attempt) so `flyctl logs` aggregation is trivial.

       Local smoke: `make dns-baseline-probe`
       Production cron (future Polywatch plan): launch via Fly scheduled machines
       or a sidecar process inside polyarb-l1.
       """
       from __future__ import annotations

       import json
       import socket
       import sys
       import time
       from datetime import datetime, timezone

       HOSTS = [
           "gamma-api.polymarket.com",
           "clob.polymarket.com",
           "data-api.polymarket.com",
       ]
       N_PROBES = 30
       INTERVAL_S = 2.0


       def probe(host: str) -> dict:
           t0 = time.monotonic()
           try:
               ip = socket.gethostbyname(host)
               latency_ms = (time.monotonic() - t0) * 1000
               return {
                   "ts": datetime.now(timezone.utc).isoformat(),
                   "host": host,
                   "ok": True,
                   "ip": ip,
                   "latency_ms": round(latency_ms, 2),
               }
           except OSError as e:
               return {
                   "ts": datetime.now(timezone.utc).isoformat(),
                   "host": host,
                   "ok": False,
                   "errno": getattr(e, "errno", None),
                   "msg": str(e)[:200],
               }


       def main() -> int:
           total = 0
           failed = 0
           for _ in range(N_PROBES):
               for host in HOSTS:
                   result = probe(host)
                   print(json.dumps(result), flush=True)
                   total += 1
                   if not result["ok"]:
                       failed += 1
               time.sleep(INTERVAL_S)
           rate = failed / total if total else 0.0
           summary = {
               "summary": True,
               "total": total,
               "failed": failed,
               "failure_rate": round(rate, 4),
           }
           print(json.dumps(summary), flush=True)
           return 0 if failed == 0 else 1


       if __name__ == "__main__":
           sys.exit(main())
       ```
