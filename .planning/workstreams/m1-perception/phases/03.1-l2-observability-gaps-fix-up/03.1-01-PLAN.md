       ```python
       def get_l2_tob_last_mirror_at_s(self) -> int | None:
           """Read last successful l2 mirror push wall-clock (seconds since epoch).

           Returns None when l2_mirror_state row absent (cold start) — caller
           treats None as "never mirrored", typically maps to /health WARN.
           """
           con = sqlite3.connect(self._db_path)
           try:
               row = con.execute(
                   "SELECT last_mirror_at_s FROM l2_mirror_state WHERE id=1"
               ).fetchone()
               return int(row[0]) if row else None
           finally:
               con.close()

       def upsert_l2_tob_mirror_state(self, last_mirror_at_s: int) -> None:
           """Write singleton row updating last successful mirror wall-clock."""
           con = sqlite3.connect(self._db_path, isolation_level=None)
           try:
               con.execute(
                   "INSERT INTO l2_mirror_state(id,last_mirror_at_s) VALUES(1,?) "
                   "ON CONFLICT(id) DO UPDATE SET last_mirror_at_s=excluded.last_mirror_at_s",
                   (int(last_mirror_at_s),),
               )
           finally:
               con.close()
       ```
