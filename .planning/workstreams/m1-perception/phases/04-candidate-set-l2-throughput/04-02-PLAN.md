```python
import os, sqlite3, tempfile
from pathlib import Path
from loguru import logger
from polyarb.storage.schemas import DDL   # authoritative full DDL (markets + aux tables)

# narrow markets_latest column → markets DDL column
_NARROW_TO_MARKETS = {
    "market_id":"market_id","question":"question","slug":"slug",
    "event_slug":"event_id","mid_price":"mid_price","liquidity_usd":"liquidity_usd",
    "volume_usd":"volume_usd","end_time_ms":"end_time_ms","snapshot_id":"snapshot_id",
    "yes_token_id":"yes_token_id",   # D-07 (Plan 01)
    # question_zh handled separately (not a markets column — inject via question_translations or skip)
}
# present in markets DDL, absent from narrow → NULL-filled (REAL nullable cols)
_NULL_FILLED_COLS = frozenset([
    "no_token_id","best_bid_price","best_bid_size","best_ask_price","best_ask_size",
    "active","closed","neg_risk","neg_risk_market_id","page_fetched_at_ms",
])
# NOT-NULL cols in DDL absent from narrow → sentinel-filled (NOT NULL)
_SENTINEL_FILL = {"condition_id": "", "fetched_at_ms": 0, "snapshot_id": 0, "incomplete": 0}

def build_temp_db(markets_rows: list[dict]) -> Path:
    """Build a NAMED-TEMP-FILE SQLite (NOT :memory:) from markets_latest rows.
    Caller MUST os.unlink(path) after use (try/finally). RESEARCH Pitfall 1."""
    with tempfile.NamedTemporaryFile(suffix=".db", delete=False) as f:
        tmp_path = Path(f.name)
    con = sqlite3.connect(tmp_path)
    try:
        con.executescript(DDL)   # creates markets + event_tags + validation_issues + question_translations
        for row in markets_rows:
            _insert_narrow_row(con, row)
        con.commit()
    finally:
        con.close()
    return tmp_path

def _insert_narrow_row(con, narrow: dict) -> None:
    mapped = {}
    for nk, mk in _NARROW_TO_MARKETS.items():
        v = narrow.get(nk)
        if nk == "event_slug":
            v = narrow.get("event_slug") or narrow.get("event_id")
        mapped[mk] = v
    # sentinel-fill NOT-NULL columns when absent
    for col, sentinel in _SENTINEL_FILL.items():
        mapped.setdefault(col, sentinel if mapped.get(col) is None else mapped[col])
        if mapped.get(col) is None:
            mapped[col] = sentinel
    cols = ",".join(mapped.keys())
    ph = ",".join("?" * len(mapped))
    con.execute(f"INSERT INTO markets ({cols}) VALUES ({ph})", list(mapped.values()))

def warn_null_filled_recipe_columns(recipe) -> None:
    """WARN (do NOT raise) when recipe WHERE/ORDER BY references a NULL-filled col.
    NULL-filled cols → 0-row results (acceptable). Missing TABLES would crash — but
    build_temp_db always creates validation_issues + event_tags, so no SQL error."""
    text = f"{recipe.where} {recipe.order_by}"
    for col in _NULL_FILLED_COLS:
        if col in text:
            logger.warning(f"recipe {recipe.name!r} uses NULL-filled column {col!r} in temp DB — will return 0 rows (markets_latest has no {col!r})")
```

CRITICAL NULL-fill correctness: the DDL has `condition_id TEXT NOT NULL`, `fetched_at_ms INTEGER NOT NULL`, `snapshot_id INTEGER NOT NULL`. These CANNOT be NULL or INSERT raises a constraint violation. Use the `_SENTINEL_FILL` sentinels ("" / 0). The TRUE-nullable columns in `_NULL_FILLED_COLS` are simply omitted (SQLite fills NULL).

**[VERIFIED FACT — DDL + FK handling, do NOT re-derive]** `from polyarb.storage.schemas import DDL` is a single multi-table string (schemas.py:43). It (a) starts with `PRAGMA foreign_keys = ON;` and (b) creates ALL tables incl. `snapshots`, `events`, `event_tags`, `markets`, `validation_issues`, `question_translations`. `markets.snapshot_id INTEGER NOT NULL REFERENCES snapshots(id)` — and because the DDL turns FK enforcement ON, inserting a markets row with a snapshot_id that has no matching snapshots row WILL raise a FK violation.

Two correct options (pick ONE, document which in SUMMARY):
  - **Option A (preferred — disable FK in the temp DB):** after `con.executescript(DDL)`, run `con.execute("PRAGMA foreign_keys = OFF")` on the build connection. The temp DB is a throwaway read-only-for-scanner artifact; FK integrity adds no value here and the scanner opens it `mode=ro` anyway. Then the snapshot_id sentinel (0) inserts cleanly. This is the simplest and is recommended.
  - **Option B (seed snapshots rows):** before inserting markets, INSERT the distinct snapshot_id values found in markets_rows into the `snapshots` table (minimal required columns). More code; only choose if FK-ON behavior is desired for some test.

Use Option A unless a test requires FK enforcement. Either way, `con.executescript(DDL)` is used directly (no DDL stripping) so scanner sees the exact production schema. Document the chosen option in the SUMMARY.
