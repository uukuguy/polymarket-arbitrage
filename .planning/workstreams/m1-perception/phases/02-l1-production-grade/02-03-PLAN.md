    4. `src/polyarb/storage/r2_sync.py`:
       ```python
       import boto3
       from botocore.config import Config
       from datetime import datetime, timezone
       from pathlib import Path
       from loguru import logger

       class R2UploadError(Exception):
           pass

       _R2_RETRY_CONFIG = Config(retries={'max_attempts': 3, 'mode': 'standard'})

       def _build_client(endpoint: str, access_key: str, secret_key: str):
           return boto3.client(
               's3', endpoint_url=endpoint,
               aws_access_key_id=access_key, aws_secret_access_key=secret_key,
               region_name='auto', config=_R2_RETRY_CONFIG,
           )

       def compute_r2_key(taken_at_ms: int) -> str:
           dt = datetime.fromtimestamp(taken_at_ms / 1000, tz=timezone.utc)
           return f"{dt:%Y}/{dt:%m}/{dt:%d}/{dt:%H}-{dt:%M}-{dt:%S}.parquet"

       def upload_parquet_to_r2(*, parquet_path: Path, bucket: str, key: str, endpoint: str, access_key: str, secret_key: str) -> str:
           client = _build_client(endpoint, access_key, secret_key)
           try:
               client.upload_file(Filename=str(parquet_path), Bucket=bucket, Key=key)
           except Exception as e:
               raise R2UploadError(f"R2 upload failed key={key}: {str(e)[:200]}") from e
           return f"{endpoint.rstrip('/')}/{bucket}/{key}"
       ```
       Note: `compute_r2_key` takes ONLY `taken_at_ms: int` (no user input) — T-02-04 mitigation per PATTERNS §2.7.
