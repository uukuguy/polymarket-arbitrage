       """Soak monitor: poll /health every interval, log to SOAK-LOG.md, alert on regression.

       Run as background script during Phase 02 soak gate:
           uv run python scripts/soak_monitor.py start --interval 300 --duration 7d
       """
       import typer, httpx, time, json
       from pathlib import Path
       from datetime import datetime, timezone, timedelta

       app = typer.Typer(no_args_is_help=True)
       LOG_PATH = Path(".planning/workstreams/m1-perception/phases/02-l1-production-grade/02-SOAK-LOG.md")

       @app.command()
       def start(interval: int = 300, duration: str = "7d", url: str = "https://polyarb-l1.fly.dev/health") -> None:
           dur_seconds = _parse_duration(duration)
           end_at = time.time() + dur_seconds
           samples: list[dict] = []
           with LOG_PATH.open("a") as log:
               log.write(f"\n## Soak run started {datetime.now(timezone.utc).isoformat()}\n")
               log.write(f"- URL: {url}\n- Interval: {interval}s\n- Duration: {duration}\n\n")
           prev_status = None
           while time.time() < end_at:
               sample = _probe(url)
               samples.append(sample)
               if sample["status"] != prev_status:
                   _log_event(log, sample, transition=True)
                   prev_status = sample["status"]
               else:
                   _log_event(log, sample, transition=False)
               time.sleep(interval)
           _write_summary(log, samples)

       def _probe(url: str) -> dict:
           try:
               t0 = time.time()
               r = httpx.get(url, timeout=10.0)
               return {"ts": datetime.now(timezone.utc).isoformat(), "http": r.status_code, "status": r.json().get("status"), "latency_ms": int((time.time()-t0)*1000)}
           except Exception as e:
               return {"ts": datetime.now(timezone.utc).isoformat(), "http": 0, "status": "unreachable", "error": str(e)[:100]}

       def _parse_duration(s: str) -> int:
           if s.endswith("d"): return int(s[:-1]) * 86400
           if s.endswith("h"): return int(s[:-1]) * 3600
           return int(s)

       @app.command()
       def status() -> None:
           """Print current soak progress summary."""

           ...

       if __name__ == "__main__":
           app()
