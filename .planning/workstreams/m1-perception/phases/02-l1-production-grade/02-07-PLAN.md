       ```python
       """Soak audit tool: pulls Better Stack 7-day history, summarizes uptime/incidents.

       This script does NOT run continuously — Better Stack monitor (Plan 05) is the
       real 7×24 prober (云端). This is a one-shot status / export tool.

       Usage:
           uv run python scripts/soak_monitor.py status     # print current uptime % + incident count
           uv run python scripts/soak_monitor.py export     # dump 7-day history to SOAK-LOG.md
       """
       import typer, httpx, os
       from pathlib import Path
       from datetime import datetime, timezone, timedelta

       app = typer.Typer(no_args_is_help=True)
       LOG_PATH = Path(".planning/workstreams/m1-perception/phases/02-l1-production-grade/02-SOAK-LOG.md")
       BS_API = "https://uptime.betterstack.com/api/v2"

       def _headers() -> dict:
           token = os.environ.get("BETTERSTACK_API_TOKEN")
           if not token:
               raise typer.Exit("BETTERSTACK_API_TOKEN env var required")
           return {"Authorization": f"Bearer {token}"}

       def _fetch_history(monitor_id: str, days: int = 7) -> dict:
           from_dt = (datetime.now(timezone.utc) - timedelta(days=days)).isoformat()
           url = f"{BS_API}/monitors/{monitor_id}/sla?from={from_dt}"
           r = httpx.get(url, headers=_headers(), timeout=30.0)
           r.raise_for_status()
           return r.json()

       @app.command()
       def status(monitor_id: str = typer.Option(..., envvar="BETTERSTACK_MONITOR_ID")) -> None:
           """Print 7-day uptime % + incident count from Better Stack."""
           h = _fetch_history(monitor_id, 7)
           availability = h.get("data", {}).get("attributes", {}).get("availability", 0)
           incidents = h.get("data", {}).get("attributes", {}).get("total_incidents", 0)
           print(f"7-day uptime: {availability:.3f}%  incidents: {incidents}")
           print(f"Gate (thread §1 生产级判定标准): uptime ≥ 99% AND incidents == 0 (or all self-healed)")
           typer.Exit(0 if availability >= 99.0 else 1)

       @app.command()
       def export(monitor_id: str = typer.Option(..., envvar="BETTERSTACK_MONITOR_ID"),
                  days: int = 7) -> None:
           """Dump Better Stack history to SOAK-LOG.md as audit trail."""
           h = _fetch_history(monitor_id, days)
           with LOG_PATH.open("a") as log:
               log.write(f"\n## Soak audit export {datetime.now(timezone.utc).isoformat()}\n")
               log.write(f"- Window: last {days} days\n")
               log.write(f"- Uptime: {h.get('data', {}).get('attributes', {}).get('availability', 0):.3f}%\n")
               log.write(f"- Incidents: {h.get('data', {}).get('attributes', {}).get('total_incidents', 0)}\n")
               log.write(f"- Raw: {h}\n\n")
           print(f"Wrote {days}-day audit to {LOG_PATH}")

       if __name__ == "__main__":
           app()
       ```
