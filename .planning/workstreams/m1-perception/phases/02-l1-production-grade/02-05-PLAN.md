    3. **src/polyarb/observability/sentry.py** — implement init_sentry:
       ```python
       import re
       import sentry_sdk
       from sentry_sdk.integrations.loguru import LoguruIntegration
       from loguru import logger
       from polyarb.config import Settings

       _SECRET_PATTERNS = [
           re.compile(r'Bearer\s+([A-Za-z0-9\-_\.]+)'),
           re.compile(r'(?:token|secret|key|api_key|access_key)\s*[=:]\s*"?([A-Za-z0-9\-_\.]+)"?', re.IGNORECASE),
           re.compile(r'sk-[A-Za-z0-9]{20,}'),
           re.compile(r'(eyJ[A-Za-z0-9_\-]{20,}\.[A-Za-z0-9_\-]{20,}\.[A-Za-z0-9_\-]{20,})'),  # JWT
       ]

       def _redact_string(s: str) -> str:
           if not isinstance(s, str):
               return s
           for pat in _SECRET_PATTERNS:
               s = pat.sub(lambda m: m.group(0).replace(m.group(1) if m.lastindex else m.group(0), "[REDACTED]"), s)
           return s

       def _redact_obj(obj):
           if isinstance(obj, str):
               return _redact_string(obj)
           if isinstance(obj, dict):
               return {k: ("[REDACTED]" if str(k).lower() in {"api_key", "token", "secret", "service_key", "access_key", "secret_key", "telegram_bot_token", "sentry_dsn", "scan_shared_secret"} else _redact_obj(v)) for k, v in obj.items()}
           if isinstance(obj, (list, tuple)):
               return type(obj)(_redact_obj(x) for x in obj)
           return obj

       def _before_send(event, hint):
           # Strip request body / extra / breadcrumb messages
           for key in ("request", "extra", "contexts", "tags"):
               if key in event:
                   event[key] = _redact_obj(event[key])
           if "breadcrumbs" in event and isinstance(event["breadcrumbs"], dict):
               values = event["breadcrumbs"].get("values", [])
               for crumb in values:
                   if "message" in crumb:
                       crumb["message"] = _redact_string(crumb["message"])
                   if "data" in crumb:
                       crumb["data"] = _redact_obj(crumb["data"])
           return event

       def init_sentry(settings: Settings) -> None:
           if not settings.sentry_dsn:
               logger.warning("sentry skipped — POLYARB_SENTRY_DSN not set")
               return
           sentry_sdk.init(
               dsn=settings.sentry_dsn,
               release=settings.release_id,
               environment="prod" if "fly.dev" in (settings.app_hostname or "") else "dev",
               send_default_pii=False,
               traces_sample_rate=0.0,  # tracing not in scope for L1 daemon
               integrations=[LoguruIntegration(level="ERROR", event_level="ERROR")],
               before_send=_before_send,
           )
           logger.info(f"sentry initialized — release={settings.release_id}")
       ```

    5. **src/polyarb/daemon/alerts.py** — implement:
       ```python
       import time
       import httpx
       import sentry_sdk
       from loguru import logger
       from polyarb.config import Settings

       _LAST_ALERT_TIME_MS: dict[str, int] = {}

       def _is_deduped(key: str, window_seconds: int) -> bool:
           now_ms = int(time.time() * 1000)
           last = _LAST_ALERT_TIME_MS.get(key)
           if last and (now_ms - last) < window_seconds * 1000:
               return True
           _LAST_ALERT_TIME_MS[key] = now_ms
           return False

       async def send_paused_alert(settings: Settings, *, reason: str) -> None:
           key = "scheduler-paused"
           if _is_deduped(key, settings.alert_dedupe_window_seconds):
               logger.debug(f"alert {key} deduplicated; not sending")
               return
           # 1. Sentry capture (long-lived breadcrumb trail)
           sentry_sdk.capture_message(f"scheduler paused: {reason}", level="error")
           # 2. Better Stack heartbeat /fail
           ok = await _better_stack_fail(settings, reason=reason)
           # 3. Fallback to direct Telegram if Better Stack unreachable
           if not ok and settings.telegram_bot_token.get_secret_value():
               await _telegram_direct(settings, text=f"polyarb-l1 scheduler PAUSED: {reason}")

       async def send_heartbeat_ok(settings: Settings) -> None:
           if not settings.better_stack_heartbeat_url:
               return
           async with httpx.AsyncClient(timeout=5.0) as client:
               try:
                   await client.get(settings.better_stack_heartbeat_url)
               except Exception as e:
                   logger.warning(f"heartbeat send failed: {e}")

       async def _better_stack_fail(settings: Settings, *, reason: str) -> bool:
           if not settings.better_stack_heartbeat_url:
               return False
           fail_url = settings.better_stack_heartbeat_url + "/fail"
           async with httpx.AsyncClient(timeout=5.0) as client:
               try:
                   resp = await client.post(fail_url, json={"reason": reason})
                   return resp.status_code < 500
               except Exception as e:
                   logger.error(f"Better Stack /fail post failed: {e}")
                   return False

       async def _telegram_direct(settings: Settings, *, text: str) -> None:
           if not settings.telegram_bot_token.get_secret_value() or not settings.telegram_chat_id:
               return
           url = f"https://api.telegram.org/bot{settings.telegram_bot_token.get_secret_value()}/sendMessage"
           async with httpx.AsyncClient(timeout=5.0) as client:
               try:
                   await client.post(url, json={"chat_id": settings.telegram_chat_id, "text": text, "parse_mode": "Markdown"})
               except Exception as e:
                   logger.error(f"Telegram direct send failed: {e}")
       ```
