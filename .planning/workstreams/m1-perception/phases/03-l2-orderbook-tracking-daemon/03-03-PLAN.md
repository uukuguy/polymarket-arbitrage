async def main() -> int:
    init_logging()                                    # 1. FIRST — sets up JSON sink
    settings = load_settings()                        # 2. config
    init_sentry(settings)                             # 3. AFTER logging (Loguru integ)
    logger.info("polyarb daemon starting up")

    sqlite_store = SQLiteStore(settings.db_path)
    sqlite_store.init_schema()

    scheduler = SnapshotScheduler(settings=settings, sqlite_store=sqlite_store)
    app = create_app(scheduler=scheduler, sqlite_store=sqlite_store, settings=settings)

    config = uvicorn.Config(app, host="0.0.0.0", port=settings.http_port, log_config=None,
                            access_log=False, loop="asyncio")
    server = uvicorn.Server(config)

    stop_event = asyncio.Event()
    loop = asyncio.get_running_loop()
    for sig in (signal.SIGINT, signal.SIGTERM):
        loop.add_signal_handler(sig, lambda s=sig: stop_event.set())

    server_task = asyncio.create_task(server.serve())

    # P9: WAIT for uvicorn socket bind BEFORE starting scheduler.
    for _ in range(100):
        if server.started:
            break
        await asyncio.sleep(0.1)
    logger.info(f"daemon running: http server on :{settings.http_port}, starting scheduler")

    scheduler_task = asyncio.create_task(scheduler.run(stop_event))
    await stop_event.wait()
    server.should_exit = True
    scheduler_task.cancel()
    try:
        await asyncio.wait_for(server_task, timeout=5.0)
    except asyncio.TimeoutError:
        ...

    def _severity(a: str, b: str) -> str:
        order = {"pass": 0, "warn": 1, "fail": 2}
        return a if order[a] >= order[b] else b


    def _build_l2_health_checks(
        store: Any,
        settings: Any,
        ws_consumer: Any | None,
        event_listener: Any | None,
        now_s: float,
    ) -> tuple[dict[str, list[dict]], str]:

        Sub-checks (sub-set may be None at Plan 03 boundary):
        - ws:connection_state (Plan 04 wires)
        - ws:last_event_age_seconds (Plan 04 wires)
        - event_bus:listener_state (Plan 05 wires)
        - mirror:l2_tob_age_seconds (Plan 06 wires)
        """
        checks: dict[str, list[dict]] = {}
        overall = "pass"

        # ── Check 1: ws:connection_state ───────────────────────────────────────
        if ws_consumer is None:
            checks["ws:connection_state"] = [{
                "status": "warn",
                "observedValue": "not_configured",
                "componentType": "websocket",
                "output": "ws_consumer not yet wired (Plan 04 deliverable)",
            }]
            overall = _severity(overall, "warn")
        else:
            state = getattr(ws_consumer, "current_state", "UNKNOWN")
            if state == "CONNECTED":
                ws_state_status = "pass"
            elif state in ("WAITING_FOR_EVENT", "RECONNECTING"):
                # Read elapsed for severity escalation
                last_at = getattr(ws_consumer, "last_event_at_s", now_s)
                age = now_s - last_at
                if state == "RECONNECTING" and age > 60:
                    ws_state_status = "fail"
                else:
                    ws_state_status = "warn"
            else:
                ws_state_status = "fail"
            checks["ws:connection_state"] = [{
                "status": ws_state_status,
                "observedValue": state,
                "componentType": "websocket",
            }]
            overall = _severity(overall, ws_state_status)

        # ── Check 2: ws:last_event_age_seconds ─────────────────────────────────
        if ws_consumer is not None:
            last_at = getattr(ws_consumer, "last_event_at_s", None)
            if last_at is None:
                age_status = "warn"
                age = None
            else:
                age = now_s - last_at
                if age < 30:
                    age_status = "pass"
                elif age < 120:
                    age_status = "warn"
                else:
                    age_status = "fail"
            checks["ws:last_event_age_seconds"] = [{
                "status": age_status,
                "observedValue": round(age, 1) if age is not None else None,
                "observedUnit": "s",
                "componentType": "websocket",
            }]
            overall = _severity(overall, age_status)

        if event_listener is None:
            checks["event_bus:listener_state"] = [{
                "status": "warn",
                "observedValue": "not_configured",
                "componentType": "asyncpg-listener",
                "output": "event_listener not yet wired (Plan 05 deliverable)",
            }]
            overall = _severity(overall, "warn")
        else:
            is_listening = getattr(event_listener, "is_listening", False)
            checks["event_bus:listener_state"] = [{
                "status": "pass" if is_listening else "warn",
                "observedValue": "listening" if is_listening else "reconnecting",
                "componentType": "asyncpg-listener",
            }]
            overall = _severity(overall, "pass" if is_listening else "warn")

        # ── Check 4: mirror:l2_tob_age_seconds (Plan 06 wires real SQLite read) ─
        if getattr(settings, "l2_mirror_enabled", False):
            try:
                # Plan 06 will populate this; Plan 03 placeholder returns None
                last_mirror_at = store.get_l2_tob_last_mirror_at_s() if hasattr(store, "get_l2_tob_last_mirror_at_s") else None
                if last_mirror_at is None:
                    mirror_status = "warn"
                    mirror_age = None
                else:
                    mirror_age = now_s - last_mirror_at
                    if mirror_age < 300:
                        mirror_status = "pass"
                    elif mirror_age < 1800:
                        mirror_status = "warn"
                    else:
                        mirror_status = "fail"
            except Exception as e:
                logger.warning(f"mirror age check failed (fail-soft): {e!r}")
                mirror_status = "warn"
                mirror_age = None

                "status": mirror_status,
                "observedValue": round(mirror_age, 1) if mirror_age is not None else None,
                "observedUnit": "s",
                "componentType": "supabase-mirror",
            }]

            "version": getattr(settings, "version", "0.0.0"),

            "serviceId": "polyarb-l2",
            "description": "Polymarket L2 orderbook tracking daemon — WS market channel + event bus",
            "checks": checks,
        }

        """IETF draft-inadarei-api-health-check-06 strict — 503 if overall=fail."""
        store = request.app.state.sqlite_store
        settings = request.app.state.settings
        ws_consumer = getattr(request.app.state, "ws_consumer", None)
        event_listener = getattr(request.app.state, "event_listener", None)
        checks, overall = _build_l2_health_checks(store, settings, ws_consumer, event_listener, time.time())
        body = _build_body(overall, checks, settings)
        http_status = 503 if overall == "fail" else 200
        return JSONResponse(body, status_code=http_status, media_type=HEALTH_CONTENT_TYPE)


    async def healthz(request: Request) -> JSONResponse:
        """Always 200 — Fly proxy reads this to avoid traffic-routing storm (Phase 02.1 BUG-6)."""

        body = _build_body(overall, checks, settings)
        return JSONResponse(body, status_code=200, media_type=HEALTH_CONTENT_TYPE)
