    def _fetch_latest_tob_rows_from_supabase(client) -> list[dict]:

    def _build_tob_temp_db(rows: list[dict]) -> Path:
        """Write l2_top_of_book rows to a named-temp SQLite for scanner.

        Blocker #2 — `ts` stored as INTEGER epoch ms so the recipe predicate
        `ts > (strftime('%s','now','-1 hour') * 1000)` works correctly.
        ISO TEXT lex comparison would mis-order across timezone or `T`
        format variations.
        """
        tmp = tempfile.NamedTemporaryFile(suffix=".sqlite", delete=False)
        tmp.close()
        path = Path(tmp.name)
        conn = sqlite3.connect(str(path))
        try:
            conn.execute("""
                CREATE TABLE l2_top_of_book (
                    asset_id TEXT,
                    ts INTEGER,           -- epoch milliseconds
                    best_bid REAL,
                    best_ask REAL,
                    spread REAL,
                    mid_price REAL,
                    depth_yes_usd REAL,
                    depth_no_usd REAL
                );
            """)

            def _iso_to_epoch_ms(ts_val) -> int | None:
                if ts_val is None: return None
                if isinstance(ts_val, (int, float)):
                    # already epoch (assume ms if > 1e12, else seconds)
                    return int(ts_val) if ts_val > 1e12 else int(ts_val * 1000)
                if isinstance(ts_val, str):
                    try:
                        s = ts_val.replace("Z", "+00:00")
                        dt = datetime.fromisoformat(s)
                        if dt.tzinfo is None:
                            dt = dt.replace(tzinfo=timezone.utc)
                        return int(dt.timestamp() * 1000)
                    except (ValueError, TypeError):
                        return None
                return None

            cols = ("asset_id", "ts", "best_bid", "best_ask", "spread",
                    "mid_price", "depth_yes_usd", "depth_no_usd")
            normalized_rows = []
            for r in rows:
                row_vals = []
                for c in cols:
                    v = r.get(c)
                    if c == "ts":
                        v = _iso_to_epoch_ms(v)
                    row_vals.append(v)
                normalized_rows.append(tuple(row_vals))
            conn.executemany(
                f"INSERT INTO l2_top_of_book ({','.join(cols)}) "
                f"VALUES ({','.join(['?'] * len(cols))})",
                normalized_rows,
            )
            conn.commit()
        finally:
            conn.close()
        return path

            except Exception as e:  # noqa: BLE001

        logger.info("l3-promote: run_periodic stopped")
