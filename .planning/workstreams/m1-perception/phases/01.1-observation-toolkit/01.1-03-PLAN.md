## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| config/scan_recipes.yaml → scanner | 用户从论坛/AI 复制陌生 yaml 是真实威胁（RESEARCH §2.1） |

| scanner output → terminal/parquet | LEFT JOIN 来的 question_zh 是 LLM 输出，可能含 ANSI escape 序列 |
| builtin recipes → scanner | 项目源码内嵌的配方（**信任**，因为是 git 受控代码） |

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|

       (d) **校验函数（带 trusted 参数）— Blocker #2 决议**：
       ```python
       def _validate_where(where: str, *, trusted: bool) -> None:
           """Validate WHERE clause string.
           trusted=True (builtin): no-op (caller authored this in source).
           trusted=False (yaml): strict _FORBIDDEN blacklist.
           """
           if trusted:
               return
           if _FORBIDDEN.search(where):
               raise ValueError(f"forbidden token in where: {where!r}")

       def _validate_order_by(order_by: str, *, trusted: bool) -> None:
           """Validate ORDER BY clause string.
           trusted=True (builtin): no-op (allows arithmetic / functions / parens).
           trusted=False (yaml): _FORBIDDEN + _ORDER_BY_OK strict whitelist
                                 (only bare columns + optional ASC/DESC).
           """
           if trusted:

       import yaml
       from loguru import logger

       def load_yaml_recipes(yaml_path: Path) -> dict[str, Recipe]:
           if not yaml_path.exists():
               return {}
           with yaml_path.open() as f:
               data = yaml.safe_load(f) or {}  # safe_load 关键
           recipes_raw = data.get("recipes", {})
           result: dict[str, Recipe] = {}
           for name, body in recipes_raw.items():
               if "group_by" in body:
                   logger.warning(
                       f"yaml recipe {name!r}: group_by is builtin-only, dropping key"
                   )
                   body = {k: v for k, v in body.items() if k != "group_by"}
               r = Recipe.from_yaml(name, body)  # _is_trusted=False
               # Validate at load time (fail-fast) — uses strict validators
               _validate_where(r.where, trusted=False)
               _validate_order_by(r.order_by, trusted=False)
               _validate_limit(r.limit)
               result[name] = r
           return result

       def list_all_recipes(yaml_path: Path | None = None) -> dict[str, Recipe]:
           merged = dict(BUILTIN_RECIPES)
           if yaml_path:
               for name, r in load_yaml_recipes(yaml_path).items():
                   if name in BUILTIN_RECIPES:
                       logger.warning(
                           f"yaml recipe {name!r} cannot override builtin, ignoring"
                       )
                       continue
                   merged[name] = r
           return merged
       ```

       ```python
       """Output formatters for observation results.

       Two output forms (CONTEXT.md `### 输出格式`):
       - A: rich.Table to terminal (default)
       - C: atomic parquet to data/scans/<recipe>/<timestamp>.parquet (always)

       Defense: rich.Table escapes markup by default (no_wrap=False, markup=False
       on add_row) — protects against ANSI escape injection via question_zh.
       """
       from __future__ import annotations
       import os
       from datetime import datetime, timezone
       from pathlib import Path
       import pandas as pd
       from loguru import logger
       from rich.console import Console
       from rich.table import Table

       _DEFAULT_COLUMNS = (
           "slug", "category", "question", "question_zh", "mid_price",
           "best_bid_price", "best_ask_price", "liquidity_usd", "volume_usd",
       )

       def render_table(df: pd.DataFrame, title: str, columns: tuple[str, ...] | None = None) -> None:
           if df.empty:
               Console().print(f"[yellow]{title}: (no rows)[/yellow]")
               return
           cols = list(columns) if columns else [c for c in _DEFAULT_COLUMNS if c in df.columns]
           if not cols:
               cols = list(df.columns)
           table = Table(title=title, show_lines=False)
           for c in cols:
               table.add_column(c, overflow="fold", no_wrap=False)
           for _, row in df.iterrows():
               # 关键：markup=False 防 question_zh 含 "[red]xxx[/red]" 渲染
               table.add_row(*[str(row.get(c, "")) for c in cols])
           Console().print(table)


       def write_scan_parquet(df: pd.DataFrame, recipe_name: str, scans_root: Path) -> Path | None:
           """Atomic write to data/scans/<recipe>/<timestamp>.parquet."""
           if df.empty:
               logger.info(f"scan {recipe_name}: 0 rows, skipping parquet write")
               return None
           timestamp = datetime.now(timezone.utc).strftime("%Y-%m-%dT%H-%M-%S")
           out_path = scans_root / recipe_name / f"{timestamp}.parquet"
           out_path.parent.mkdir(parents=True, exist_ok=True)
           tmp = out_path.with_suffix(".parquet.tmp")
           try:
               df.to_parquet(tmp, compression="snappy", index=False)
           except Exception:
               tmp.unlink(missing_ok=True)
               raise
           os.replace(tmp, out_path)
           logger.info(f"scan {recipe_name}: wrote {len(df)} rows → {out_path}")
           return out_path
       ```
