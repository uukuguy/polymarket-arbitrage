#### 长生命客户端 + async context manager (gamma_client.py:55-89)

```python
class GammaClient:
    def __init__(self, settings: Settings) -> None:
        self._settings = settings
        self._limiter = AsyncLimiter(settings.gamma_rate_per_10s, 10)
        self._http = httpx.AsyncClient(
            timeout=settings.http_timeout_s,
            limits=httpx.Limits(max_connections=20, max_keepalive_connections=10),
            headers={"User-Agent": "polyarb/0.1"},
            http2=True,
            follow_redirects=False,  # F-2 SECURITY
        )

    async def aclose(self) -> None:
        await self._http.aclose()

    async def __aenter__(self) -> GammaClient:
        return self

    async def __aexit__(self, exc_type, exc, tb) -> None:
        await self.aclose()
```

新 `TranslationClient` 直接套用：
- `__init__` 持 `AsyncOpenAI` + `asyncio.Semaphore`，**无网络 IO**
- `aclose()` 关 client（idempotent）
- `__aenter__` / `__aexit__` 配对（用法：`async with TranslationClient(...) as t: ...`）

⚠ **Pitfall 6 from RESEARCH.md**：orchestrator 末尾必须 `await translator.aclose()`，否则 event loop 退出 warning。

```python
def _phase(label: str):
    """Bracket a pipeline phase with start/done log lines and elapsed timing."""
    logger.info(f"Phase {label} — start")
    t0 = time.monotonic()
```

**Analog:** `src/polyarb/storage/sqlite_store.py`

#### Imports + `_BOOL_COLUMNS` 模板 (sqlite_store.py:1-31)

```python

Per CONTEXT.md D-D3, write_snapshot persists the row even when is_valid=False so

- Uses stdlib sqlite3 + parameterized DDL/SQL — NO SQLAlchemy ORM.
- BEGIN IMMEDIATE + DELETE FROM markets + executemany INSERT — never INSERT OR
```

#### init_schema 模式 (sqlite_store.py:69-75)

```python
def init_schema(self) -> None:
    """Create tables, indexes, set WAL mode. Idempotent — safe to re-run."""
    con = sqlite3.connect(self._db_path, isolation_level=None)
    try:
        con.executescript(DDL)
    finally:
        con.close()
```

`question_translations` 表的 DDL **直接追加到 `schemas.py:DDL` 字符串**（不创建新 SQLiteStore）。`SQLiteStore.init_schema()` 已经能 idempotent 建表，cache.py **只负责 CRUD**，不另起一套 init。

#### 单连接事务 (sqlite_store.py:100-154)

```python

con.execute("PRAGMA journal_mode=WAL")
con.execute("PRAGMA synchronous=NORMAL")
con.execute("BEGIN IMMEDIATE")
try:
    ...
    con.executemany(MARKETS_INSERT_SQL, market_tuples)
    con.execute("COMMIT")
except Exception:
    con.execute("ROLLBACK")
    logger.exception("SQLite write_snapshot rolled back")

finally:
    con.close()
```

- `BEGIN IMMEDIATE` 包整批 insert
- 用 `INSERT OR IGNORE INTO question_translations(...)` 或 `ON CONFLICT(question_hash) DO UPDATE` 处理重复 question
- 异常 → ROLLBACK + raise（不静默吞）

#### Atomic write 原型 (parquet_writer.py:41-64)

```python
def write_parquet_atomic(rows: list[dict], out_path: Path) -> None:
    """Write rows as a Parquet file at out_path, atomically.

    Caller is responsible for shaping rows to match SNAPSHOT_SCHEMA (no schema
    coercion here — pyarrow will raise on mismatch and we let it propagate).

    On any exception during write_table, the .tmp file is removed before re-raise
    so the destination directory contains no partial files.
    """
    out_path = Path(out_path)
    out_path.parent.mkdir(parents=True, exist_ok=True)

    table = pa.Table.from_pylist(rows, schema=SNAPSHOT_SCHEMA)

    tmp = out_path.with_suffix(out_path.suffix + ".tmp")
    try:
        pq.write_table(table, tmp, compression="snappy")
    except Exception:
        tmp.unlink(missing_ok=True)
        raise

    os.replace(tmp, out_path)
    logger.info(f"Parquet written: {out_path} ({len(rows)} rows)")
```

formatter.py 落 `data/scans/<scan-name>/<timestamp>.parquet` 时复用同款 tmp + os.replace 模式。**不要新建一套 schema**——scan 输出的列就是 `markets.*` + `question_zh`，直接用 pandas DataFrame `to_parquet()` 自动推导即可（不像 snapshot 必须强 schema 因为 token_id 70 位字符串风险）。

#### 完整 typer 模板 (snapshot/cli.py:1-89)

```python
"""polyarb CLI: snapshot subcommand.

D-F1 (default): single-line stdout summary.
D-F2 (--verbose): DEBUG-level loguru output to stderr.
D-F3 (failure): stderr summary on is_valid=False, exit code 1.
"""
from __future__ import annotations

import asyncio
import sys
from pathlib import Path

import typer
from loguru import logger

from polyarb.config import load_settings
from polyarb.snapshot.orchestrator import run_snapshot

app = typer.Typer(no_args_is_help=True, add_completion=False)


@app.command()
def snapshot(
    full: bool = typer.Option(False, "--full", help="..."),
    verbose: bool = typer.Option(False, "--verbose", "-v", help="..."),
    no_cache: bool = typer.Option(False, "--no-cache", help="..."),
    config: Path | None = typer.Option(None, "--config", help="..."),
) -> None:
    """Capture a one-shot Polymarket market snapshot."""
    logger.remove()
    logger.add(
        sys.stderr,
        level="DEBUG" if verbose else "INFO",
        format="<green>{time:HH:mm:ss}</green> | <level>{level:<7}</level> | {message}",
```

### 3.4 路径校验（F-3 SECURITY）

```python
@field_validator("db_path", "parquet_root", "cache_root")
@classmethod
def _within_project(cls, v: Path) -> Path:
    if os.environ.get("POLYARB_ALLOW_EXTERNAL_PATHS") == "1":
        return v.resolve() if v.is_absolute() else (Path.cwd() / v).resolve()
    project_root = Path.cwd().resolve()
    resolved = (project_root / v).resolve() if not v.is_absolute() else v.resolve()
    try:
        resolved.relative_to(project_root)
    except ValueError as e:
        raise ValueError(f"path {v} resolves outside project root {project_root}") from e
    return resolved
```
