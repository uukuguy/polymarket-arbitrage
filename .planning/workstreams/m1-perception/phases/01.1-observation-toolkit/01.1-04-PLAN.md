       ```python
       """Track a single market across all snapshot parquet files via DuckDB.

       Strategy (RESEARCH §3.1-3.6):
       - in-memory duckdb (no spill needed at our scale ~ 100 parquet × 20k rows)
       - glob across data/snapshots/**/*.parquet (POSIX-style, duckdb-native)
       - union_by_name=true future-proofs against schema additions (e.g. Phase
         1.1 adds category/tags; older parquet auto-NULL filled)
       - WHERE on slug pushes down via parquet zonemaps (RESEARCH §3.1)

       Pitfalls (RESEARCH §3.6):
       - fetchdf() materializes to pandas — fine at 20k rows, switch to .arrow()
         if scaling to 1M+
       - union_by_name only aligns names not types; if a future phase changes a
         column's type, must rewrite history (RESEARCH §3.5 invariant)

       Output column note (Warning #7 决议):
       - taken_at_ms: snapshot_taken_at_ms (秒级；同秒可能撞)
       - snapshot_id: tie-breaker for same-second snapshots (rare but real
         when re-running snapshot in close succession; ORDER BY (taken_at_ms,
         snapshot_id) is stable)
       """
       from __future__ import annotations
       from pathlib import Path
       import duckdb
       import pandas as pd
       from loguru import logger

       _PARQUET_COUNT_WARN_THRESHOLD = 200  # RESEARCH A4

       def track_market(slug: str, parquet_root: Path) -> pd.DataFrame:
           glob = str(parquet_root / "**" / "*.parquet")
           parquet_files = list(parquet_root.rglob("*.parquet"))
           if len(parquet_files) > _PARQUET_COUNT_WARN_THRESHOLD:
               logger.warning(
                   f"track_market: scanning {len(parquet_files)} parquet files "
                   f"(>{_PARQUET_COUNT_WARN_THRESHOLD}); in-memory duckdb may slow"
               )
           con = duckdb.connect()  # in-memory
           try:
               return con.execute(
                   """
                   SELECT
                       snapshot_taken_at_ms AS taken_at_ms,
                       snapshot_id,
                       slug,
                       question,
                       category,
                       mid_price,
                       best_bid_price,
                       best_ask_price,
                       (best_ask_price - best_bid_price) AS spread,
                       liquidity_usd,
                       volume_usd
                   FROM read_parquet(?, union_by_name=true)
                   WHERE slug = ?
                   ORDER BY taken_at_ms, snapshot_id
                   """,
                   [glob, slug],  # 参数化（防 T-01.1-16）
               ).fetchdf()
           finally:
               con.close()
       ```

       ```python
       """Compare two snapshots — show drift / appeared / vanished markets.

       Resolution: snapshot_id (int) → SELECT parquet_path FROM snapshots → DuckDB
       The user gives integer snapshot IDs (from SQLite snapshots.id), never raw
       paths (T-01.1-15 — defense against arbitrary file access).
       """
       from __future__ import annotations
       import sqlite3
       from pathlib import Path
       import duckdb
       import pandas as pd

       def resolve_snapshot_path(snapshot_id: int, db_path: Path) -> Path:
           if not isinstance(snapshot_id, int) or snapshot_id < 1:
               raise ValueError(f"snapshot_id must be positive int, got {snapshot_id!r}")
           con = sqlite3.connect(f"file:{db_path}?mode=ro", uri=True)
           try:
               row = con.execute(
                   "SELECT parquet_path FROM snapshots WHERE id = ?", (snapshot_id,)
               ).fetchone()
               if row is None:
                   raise ValueError(f"snapshot_id {snapshot_id} not found")
               return Path(row[0])
           finally:
               con.close()

       def latest_snapshot_pair(db_path: Path) -> tuple[int, int]:
           """Return (second-newest, newest) snapshot IDs."""
           con = sqlite3.connect(f"file:{db_path}?mode=ro", uri=True)
           try:
               rows = con.execute(
                   "SELECT id FROM snapshots ORDER BY id DESC LIMIT 2"
               ).fetchall()
               if len(rows) < 2:
                   raise ValueError("need at least 2 snapshots to compare; only "
                                    f"{len(rows)} found")
               return (rows[1][0], rows[0][0])  # (older, newer)
           finally:
               con.close()

       def compare_snapshots(from_path: Path, to_path: Path) -> pd.DataFrame:
           """Diff two specific snapshots — drift / appeared / vanished."""
           con = duckdb.connect()
           try:
               return con.execute(
                   """
                   WITH a AS (SELECT * FROM read_parquet(?, union_by_name=true)),
                        b AS (SELECT * FROM read_parquet(?, union_by_name=true))
                   SELECT
                       COALESCE(a.slug, b.slug) AS slug,
                       COALESCE(a.question, b.question) AS question,
                       COALESCE(a.category, b.category) AS category,
                       a.mid_price AS mid_from,
                       b.mid_price AS mid_to,
                       (b.mid_price - a.mid_price) AS mid_drift,
                       a.liquidity_usd AS liq_from,
                       b.liquidity_usd AS liq_to,
                       (b.liquidity_usd - a.liquidity_usd) AS liq_drift,
                       CASE
                           WHEN a.slug IS NULL THEN 'appeared'
                           WHEN b.slug IS NULL THEN 'vanished'
                           ELSE 'persistent'
                       END AS state
                   FROM a FULL OUTER JOIN b USING (slug)
                   ORDER BY ABS(COALESCE(b.mid_price, 0) - COALESCE(a.mid_price, 0)) DESC
                   """,
                   [str(from_path), str(to_path)],
               ).fetchdf()
           finally:
               con.close()
       ```
